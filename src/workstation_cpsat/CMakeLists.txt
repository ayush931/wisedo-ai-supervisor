cmake_minimum_required(VERSION 3.22)
project(supervisor_workstation_cpsat LANGUAGES CXX)
option(WORKSTATION_CPSAT_BUILD_TESTING "Build standalone scheduler tests" OFF)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
if(WORKSTATION_CPSAT_ORTOOLS_ROOT)
  list(PREPEND CMAKE_PREFIX_PATH "${WORKSTATION_CPSAT_ORTOOLS_ROOT}")
endif()
find_package(ortools 9.14 CONFIG QUIET)
if(NOT ortools_FOUND)
  # shortcut: source fallback compiles upstream OR-Tools and is slower; use a pinned binary package in CI/containers.
  set(BUILD_DEPS ON CACHE BOOL "" FORCE)
  set(BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(BUILD_FLATZINC OFF CACHE BOOL "" FORCE)
  set(BUILD_MATH_OPT OFF CACHE BOOL "" FORCE)
  set(USE_HIGHS OFF CACHE BOOL "" FORCE)
  set(USE_PDLP OFF CACHE BOOL "" FORCE)
  set(USE_SCIP OFF CACHE BOOL "" FORCE)
  set(USE_COINOR OFF CACHE BOOL "" FORCE)
  set(USE_GUROBI OFF CACHE BOOL "" FORCE)
  set(USE_XPRESS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(ortools GIT_REPOSITORY https://github.com/google/or-tools.git GIT_TAG v9.14 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(ortools)
endif()
find_package(nlohmann_json 3.11 QUIET)
if(NOT nlohmann_json_FOUND)
  FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
  FetchContent_MakeAvailable(json)
endif()

add_executable(supervisor_workstation_cpsat src/main.cpp src/solver.cpp)
set_target_properties(supervisor_workstation_cpsat PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_link_libraries(supervisor_workstation_cpsat PRIVATE ortools::ortools nlohmann_json::nlohmann_json)
target_compile_options(supervisor_workstation_cpsat PRIVATE -Wall -Wextra -Wpedantic)

include(CTest)
if(WORKSTATION_CPSAT_BUILD_TESTING)
  add_executable(workstation_cpsat_test tests/test_solver.cpp src/solver.cpp)
  target_include_directories(workstation_cpsat_test PRIVATE src)
  target_link_libraries(workstation_cpsat_test PRIVATE ortools::ortools nlohmann_json::nlohmann_json)
  target_compile_options(workstation_cpsat_test PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME workstation_cpsat_test COMMAND workstation_cpsat_test)
endif()
